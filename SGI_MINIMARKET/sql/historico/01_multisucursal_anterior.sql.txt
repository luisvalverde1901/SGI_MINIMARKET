-- SGI MINIMARKET - MIGRACIÓN MULTISUCURSAL (SPRINTS 0-2)
-- Ejecutar UNA VEZ en Supabase > SQL Editor después de realizar una copia de seguridad.
-- Idempotente en lo esencial: los inventarios existentes no se sobrescriben.
-- No borra ni renombra la columna legacy productos.stock.
-- Requiere ejecutar como administrador de la base de datos (SQL Editor).
BEGIN;

CREATE SCHEMA IF NOT EXISTS private;
REVOKE ALL ON SCHEMA private FROM PUBLIC, anon;

-- Compatibilidad con la instalación actual. Si ya existen, se conservan los datos.
CREATE TABLE IF NOT EXISTS public.perfiles (
  id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  nombre text NOT NULL,
  rol text NOT NULL DEFAULT 'empleado' CHECK (rol IN ('administrador','empleado')),
  activo boolean NOT NULL DEFAULT true
);

CREATE TABLE IF NOT EXISTS public.productos (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  codigo text NOT NULL UNIQUE,
  nombre text NOT NULL,
  categoria text NOT NULL,
  precio numeric(12,2) NOT NULL DEFAULT 0,
  stock integer NOT NULL DEFAULT 0,
  stock_minimo integer NOT NULL DEFAULT 5,
  registrado_por uuid REFERENCES auth.users(id)
);

-- En el SGI anterior estas columnas eran la fuente del stock general.
-- Se mantienen como valores históricos para evitar pérdida de información.
ALTER TABLE public.productos ADD COLUMN IF NOT EXISTS stock integer DEFAULT 0;
ALTER TABLE public.productos ADD COLUMN IF NOT EXISTS stock_minimo integer DEFAULT 5;

CREATE TABLE IF NOT EXISTS public.sucursales (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  codigo text NOT NULL UNIQUE,
  nombre text NOT NULL,
  direccion text,
  activo boolean NOT NULL DEFAULT true,
  creado_en timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.usuarios_sucursales (
  usuario_id uuid NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  sucursal_id bigint NOT NULL REFERENCES public.sucursales(id) ON DELETE CASCADE,
  rol text NOT NULL CHECK (rol IN ('administrador_sucursal','empleado')),
  activo boolean NOT NULL DEFAULT true,
  PRIMARY KEY (usuario_id, sucursal_id)
);
CREATE INDEX IF NOT EXISTS idx_usuario_sucursal_sede ON public.usuarios_sucursales (sucursal_id);

CREATE TABLE IF NOT EXISTS public.inventarios (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  producto_id bigint NOT NULL REFERENCES public.productos(id) ON DELETE RESTRICT,
  sucursal_id bigint NOT NULL REFERENCES public.sucursales(id) ON DELETE RESTRICT,
  stock integer NOT NULL DEFAULT 0 CHECK (stock >= 0),
  stock_minimo integer NOT NULL DEFAULT 5 CHECK (stock_minimo >= 0),
  actualizado_en timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT inventarios_producto_sucursal_unica UNIQUE (producto_id, sucursal_id)
);
CREATE INDEX IF NOT EXISTS idx_inventarios_sucursal ON public.inventarios(sucursal_id);

CREATE TABLE IF NOT EXISTS public.movimientos_inventario (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  producto_id bigint NOT NULL REFERENCES public.productos(id) ON DELETE RESTRICT,
  sucursal_id bigint NOT NULL REFERENCES public.sucursales(id) ON DELETE RESTRICT,
  tipo text NOT NULL CHECK (tipo IN ('entrada','salida')),
  cantidad integer NOT NULL CHECK (cantidad > 0),
  stock_anterior integer NOT NULL CHECK (stock_anterior >= 0),
  stock_nuevo integer NOT NULL CHECK (stock_nuevo >= 0),
  motivo text,
  usuario_id uuid NOT NULL REFERENCES auth.users(id) ON DELETE RESTRICT,
  creado_en timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_movimientos_sucursal_fecha
  ON public.movimientos_inventario(sucursal_id, creado_en DESC);

-- Sucursal inicial para trasladar el stock ya registrado en productos.
INSERT INTO public.sucursales (codigo,nombre,direccion)
VALUES ('PRINCIPAL','Sucursal principal','Sede inicial migrada del SGI anterior')
ON CONFLICT (codigo) DO NOTHING;

-- Atención: solo se importan las cantidades UNA VEZ por combinación producto/sucursal.
-- Los cambios futuros se hacen en inventarios, nunca en productos.stock.
INSERT INTO public.inventarios (producto_id,sucursal_id,stock,stock_minimo)
SELECT p.id, s.id, GREATEST(COALESCE(p.stock,0),0),
       GREATEST(COALESCE(p.stock_minimo,5),0)
FROM public.productos p
CROSS JOIN public.sucursales s
WHERE s.codigo='PRINCIPAL'
ON CONFLICT (producto_id,sucursal_id) DO NOTHING;

-- Mantener operativos a los empleados que ya tenían perfil en el SGI antiguo.
-- Los administradores con perfiles.rol='administrador' quedan como generales.
INSERT INTO public.usuarios_sucursales(usuario_id,sucursal_id,rol)
SELECT p.id,s.id,'empleado'
FROM public.perfiles p
CROSS JOIN public.sucursales s
WHERE s.codigo='PRINCIPAL' AND p.rol='empleado'
ON CONFLICT (usuario_id,sucursal_id) DO NOTHING;

-- Funciones de autorización en esquema privado NO expuesto por la API.
CREATE OR REPLACE FUNCTION private.es_admin_global()
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$
  SELECT auth.uid() IS NOT NULL AND EXISTS (
    SELECT 1 FROM public.perfiles p
    WHERE p.id = auth.uid() AND p.activo = true AND p.rol = 'administrador'
  );
$$;

CREATE OR REPLACE FUNCTION private.puede_ver_sucursal(p_sucursal bigint)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$
  SELECT auth.uid() IS NOT NULL AND EXISTS (
    SELECT 1 FROM public.perfiles p
    JOIN public.sucursales s ON s.id = p_sucursal AND s.activo = true
    WHERE p.id = auth.uid() AND p.activo = true AND (
      p.rol = 'administrador' OR EXISTS (
        SELECT 1 FROM public.usuarios_sucursales us
        WHERE us.usuario_id = p.id AND us.sucursal_id = p_sucursal AND us.activo
      )
    )
  );
$$;

CREATE OR REPLACE FUNCTION private.puede_gestionar_sucursal(p_sucursal bigint)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$
  SELECT auth.uid() IS NOT NULL AND EXISTS (
    SELECT 1 FROM public.perfiles p
    JOIN public.sucursales s ON s.id = p_sucursal AND s.activo = true
    WHERE p.id = auth.uid() AND p.activo = true AND (
      p.rol = 'administrador' OR EXISTS (
        SELECT 1 FROM public.usuarios_sucursales us
        WHERE us.usuario_id = p.id AND us.sucursal_id = p_sucursal
          AND us.activo AND us.rol = 'administrador_sucursal'
      )
    )
  );
$$;

CREATE OR REPLACE FUNCTION private.usuario_valido()
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$
  SELECT auth.uid() IS NOT NULL AND EXISTS (
    SELECT 1 FROM public.perfiles p
    WHERE p.id = auth.uid() AND p.activo = true AND (
      p.rol = 'administrador' OR EXISTS (
        SELECT 1 FROM public.usuarios_sucursales us
        JOIN public.sucursales s ON s.id = us.sucursal_id
        WHERE us.usuario_id = p.id AND us.activo AND s.activo
      )
    )
  );
$$;

-- La operación completa ocurre en una única transacción Postgres.
-- BLOQUEA la fila de inventario para evitar stock negativo por concurrencia.
CREATE OR REPLACE FUNCTION private.registrar_movimiento_interno(
  p_producto_id bigint, p_sucursal_id bigint, p_tipo text,
  p_cantidad integer, p_motivo text
) RETURNS integer LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$
DECLARE v_inventario_id bigint; v_anterior integer; v_nuevo integer;
BEGIN
  IF NOT private.puede_ver_sucursal(p_sucursal_id) THEN
    RAISE EXCEPTION 'No tienes acceso a esta sucursal.';
  END IF;
  IF p_tipo IS NULL OR p_tipo NOT IN ('entrada','salida')
    OR p_cantidad IS NULL OR p_cantidad <= 0 THEN
    RAISE EXCEPTION 'Tipo o cantidad de movimiento inválidos.';
  END IF;
  IF NOT EXISTS (SELECT 1 FROM public.productos WHERE id=p_producto_id) THEN
    RAISE EXCEPTION 'El producto no existe.';
  END IF;
  IF p_tipo='entrada' THEN
    INSERT INTO public.inventarios(producto_id,sucursal_id,stock,stock_minimo)
    VALUES(p_producto_id,p_sucursal_id,0,5)
    ON CONFLICT(producto_id,sucursal_id) DO NOTHING;
  END IF;
  SELECT id,stock INTO v_inventario_id,v_anterior
    FROM public.inventarios
    WHERE producto_id=p_producto_id AND sucursal_id=p_sucursal_id
    FOR UPDATE;
  IF v_inventario_id IS NULL THEN
    RAISE EXCEPTION 'No existe stock de este producto en la sucursal.';
  END IF;
  v_nuevo := CASE WHEN p_tipo='entrada' THEN v_anterior+p_cantidad
                  ELSE v_anterior-p_cantidad END;
  IF v_nuevo < 0 THEN
    RAISE EXCEPTION 'Stock insuficiente. Disponible: %.',v_anterior;
  END IF;
  UPDATE public.inventarios SET stock=v_nuevo,actualizado_en=now()
    WHERE id=v_inventario_id;
  INSERT INTO public.movimientos_inventario
    (producto_id,sucursal_id,tipo,cantidad,stock_anterior,stock_nuevo,motivo,usuario_id)
    VALUES(p_producto_id,p_sucursal_id,p_tipo,p_cantidad,v_anterior,v_nuevo,
      nullif(left(trim(coalesce(p_motivo,'')),200),''),auth.uid());
  RETURN v_nuevo;
END;
$$;

-- Creación atómica: catálogo + inventario de sucursal + entrada inicial.
CREATE OR REPLACE FUNCTION private.crear_producto_con_stock_interno(
  p_codigo text, p_nombre text, p_categoria text, p_precio numeric,
  p_sucursal_id bigint, p_stock_inicial integer, p_stock_minimo integer
) RETURNS bigint LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$
DECLARE v_producto_id bigint;
BEGIN
  IF NOT private.puede_ver_sucursal(p_sucursal_id) THEN
    RAISE EXCEPTION 'No tienes permiso para registrar en esta sucursal.';
  END IF;
  IF coalesce(length(trim(p_codigo)),0)=0 OR coalesce(length(trim(p_nombre)),0)=0
    OR coalesce(length(trim(p_categoria)),0)=0
    OR length(p_codigo)>30 OR length(p_nombre)>120
    OR p_precio IS NULL OR p_precio<0
    OR p_stock_inicial IS NULL OR p_stock_inicial<0
    OR p_stock_minimo IS NULL OR p_stock_minimo<0 THEN
    RAISE EXCEPTION 'Revisa los datos del nuevo producto.';
  END IF;
  INSERT INTO public.productos
    (codigo,nombre,categoria,precio,stock,stock_minimo,registrado_por)
    VALUES(trim(p_codigo),trim(p_nombre),trim(p_categoria),p_precio,0,p_stock_minimo,auth.uid())
    RETURNING id INTO v_producto_id;
  INSERT INTO public.inventarios(producto_id,sucursal_id,stock,stock_minimo)
    VALUES(v_producto_id,p_sucursal_id,0,p_stock_minimo);
  IF p_stock_inicial>0 THEN
    PERFORM private.registrar_movimiento_interno(
      v_producto_id,p_sucursal_id,'entrada',p_stock_inicial,'Stock inicial');
  END IF;
  RETURN v_producto_id;
END;
$$;

-- Funciones públicas INVOKER; delegan en funciones seguras del esquema privado.
CREATE OR REPLACE FUNCTION public.registrar_movimiento(
  p_producto_id bigint,p_sucursal_id bigint,p_tipo text,
  p_cantidad integer,p_motivo text DEFAULT NULL
) RETURNS integer LANGUAGE sql SECURITY INVOKER SET search_path = '' AS $$
  SELECT private.registrar_movimiento_interno($1,$2,$3,$4,$5);
$$;

CREATE OR REPLACE FUNCTION public.crear_producto_con_stock(
  p_codigo text,p_nombre text,p_categoria text,p_precio numeric,
  p_sucursal_id bigint,p_stock_inicial integer,p_stock_minimo integer
) RETURNS bigint LANGUAGE sql SECURITY INVOKER SET search_path = '' AS $$
  SELECT private.crear_producto_con_stock_interno($1,$2,$3,$4,$5,$6,$7);
$$;

-- Reemplazar TODAS las políticas previas en las tablas involucradas:
-- una política antigua demasiado amplia permitiría saltar el aislamiento de sedes.
DO $$ DECLARE p record; BEGIN
  FOR p IN SELECT schemaname,tablename,policyname FROM pg_policies
           WHERE schemaname='public'
           AND tablename IN ('perfiles','productos','sucursales',
             'usuarios_sucursales','inventarios','movimientos_inventario')
  LOOP
    EXECUTE format('DROP POLICY %I ON %I.%I',p.policyname,p.schemaname,p.tablename);
  END LOOP;
END $$;

ALTER TABLE public.perfiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.productos ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sucursales ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.usuarios_sucursales ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.inventarios ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.movimientos_inventario ENABLE ROW LEVEL SECURITY;

CREATE POLICY perfiles_lectura ON public.perfiles FOR SELECT TO authenticated
  USING (id=(SELECT auth.uid()) OR (SELECT private.es_admin_global()));

CREATE POLICY productos_lectura ON public.productos FOR SELECT TO authenticated
  USING ((SELECT private.usuario_valido()));
CREATE POLICY productos_edicion ON public.productos FOR UPDATE TO authenticated
  USING ((SELECT private.es_admin_global()))
  WITH CHECK ((SELECT private.es_admin_global()));

CREATE POLICY sedes_lectura ON public.sucursales FOR SELECT TO authenticated
  USING ((SELECT private.es_admin_global()) OR
    (activo AND (SELECT private.puede_ver_sucursal(id))));
CREATE POLICY sedes_alta ON public.sucursales FOR INSERT TO authenticated
  WITH CHECK ((SELECT private.es_admin_global()));
CREATE POLICY sedes_edicion ON public.sucursales FOR UPDATE TO authenticated
  USING ((SELECT private.es_admin_global()))
  WITH CHECK ((SELECT private.es_admin_global()));

CREATE POLICY asignaciones_lectura ON public.usuarios_sucursales FOR SELECT TO authenticated
  USING (usuario_id=(SELECT auth.uid()) OR (SELECT private.es_admin_global()));
CREATE POLICY asignaciones_alta ON public.usuarios_sucursales FOR INSERT TO authenticated
  WITH CHECK ((SELECT private.es_admin_global()));
CREATE POLICY asignaciones_edicion ON public.usuarios_sucursales FOR UPDATE TO authenticated
  USING ((SELECT private.es_admin_global()))
  WITH CHECK ((SELECT private.es_admin_global()));
CREATE POLICY asignaciones_baja ON public.usuarios_sucursales FOR DELETE TO authenticated
  USING ((SELECT private.es_admin_global()));

CREATE POLICY inventarios_lectura ON public.inventarios FOR SELECT TO authenticated
  USING (private.puede_ver_sucursal(sucursal_id));
CREATE POLICY inventarios_minimo ON public.inventarios FOR UPDATE TO authenticated
  USING (private.puede_gestionar_sucursal(sucursal_id))
  WITH CHECK (private.puede_gestionar_sucursal(sucursal_id));

CREATE POLICY movimientos_lectura ON public.movimientos_inventario FOR SELECT TO authenticated
  USING (private.puede_ver_sucursal(sucursal_id));

-- No se conceden INSERT/UPDATE de stock sobre inventarios ni INSERT de movimientos.
-- Las funciones atómicas son el único modo de cambiar stock y registrar historial.
REVOKE ALL ON public.perfiles FROM anon,authenticated;
GRANT SELECT ON public.perfiles TO authenticated;
REVOKE ALL ON public.productos FROM anon,authenticated;
GRANT SELECT,UPDATE ON public.productos TO authenticated;
REVOKE ALL ON public.sucursales FROM anon,authenticated;
GRANT SELECT,INSERT,UPDATE ON public.sucursales TO authenticated;
REVOKE ALL ON public.usuarios_sucursales FROM anon,authenticated;
GRANT SELECT,INSERT,UPDATE,DELETE ON public.usuarios_sucursales TO authenticated;
REVOKE ALL ON public.inventarios FROM anon,authenticated;
GRANT SELECT ON public.inventarios TO authenticated;
GRANT UPDATE(stock_minimo) ON public.inventarios TO authenticated;
REVOKE ALL ON public.movimientos_inventario FROM anon,authenticated;
GRANT SELECT ON public.movimientos_inventario TO authenticated;

-- Claves identity/serial para inserts autorizados.
GRANT USAGE ON SCHEMA private TO authenticated;
-- Revocar solo nuestras funciones (no alterar otras funciones privadas del proyecto).
REVOKE ALL ON FUNCTION private.es_admin_global() FROM PUBLIC,anon,authenticated;
REVOKE ALL ON FUNCTION private.puede_ver_sucursal(bigint) FROM PUBLIC,anon,authenticated;
REVOKE ALL ON FUNCTION private.puede_gestionar_sucursal(bigint) FROM PUBLIC,anon,authenticated;
REVOKE ALL ON FUNCTION private.usuario_valido() FROM PUBLIC,anon,authenticated;
REVOKE ALL ON FUNCTION private.registrar_movimiento_interno(bigint,bigint,text,integer,text) FROM PUBLIC,anon,authenticated;
REVOKE ALL ON FUNCTION private.crear_producto_con_stock_interno(text,text,text,numeric,bigint,integer,integer) FROM PUBLIC,anon,authenticated;
GRANT EXECUTE ON FUNCTION private.es_admin_global() TO authenticated;
GRANT EXECUTE ON FUNCTION private.puede_ver_sucursal(bigint) TO authenticated;
GRANT EXECUTE ON FUNCTION private.puede_gestionar_sucursal(bigint) TO authenticated;
GRANT EXECUTE ON FUNCTION private.usuario_valido() TO authenticated;
GRANT EXECUTE ON FUNCTION private.registrar_movimiento_interno(bigint,bigint,text,integer,text) TO authenticated;
GRANT EXECUTE ON FUNCTION private.crear_producto_con_stock_interno(text,text,text,numeric,bigint,integer,integer) TO authenticated;
REVOKE ALL ON FUNCTION public.registrar_movimiento(bigint,bigint,text,integer,text) FROM PUBLIC,anon;
GRANT EXECUTE ON FUNCTION public.registrar_movimiento(bigint,bigint,text,integer,text) TO authenticated;
REVOKE ALL ON FUNCTION public.crear_producto_con_stock(text,text,text,numeric,bigint,integer,integer) FROM PUBLIC,anon;
GRANT EXECUTE ON FUNCTION public.crear_producto_con_stock(text,text,text,numeric,bigint,integer,integer) TO authenticated;

COMMIT;
